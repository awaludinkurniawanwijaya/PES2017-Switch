# ARMv8.2-A baseline for the phone build: without it the NDK emits calls to the
# outline-atomics helpers for every guest lwarx/stwcx. (std::atomic CAS) instead
# of one LSE instruction. Every phone since the Snapdragon 845 / Cortex-A75 has it;
# the app checks /proc/cpuinfo before loading (NativeBridge.kt). Turn off for a
# build that must run on ARMv8.0 cores.
option(PES_ARMV82 "Build for ARMv8.2-A (LSE atomics, fp16, dotprod) on Android" ON)

    string(APPEND CMAKE_CXX_FLAGS " -march=armv8.2-a+fp16+dotprod")
endif()

        "matching pes/default.xex before configuring the full runtime target. "
        "Required: ppc_recomp.*.cpp, ppc_func_mapping.cpp, and stubs_generated.cpp.")

endif()
target_include_directories(ppc_recomp PUBLIC
    "${PES17_PPC_DIR}"
    "${SIMDE_INCLUDE_DIR}"
    "${SIMDE_INCLUDE_DIR}/simde")
# The generated code is machine output; its warnings are noise.
target_compile_options(ppc_recomp PRIVATE -w)
# PowerPC separate fmul/fadd must stay separate (fused forms are emitted
# explicitly where the guest fuses), and the generated accesses alias through
# the guest base pointer: contraction and strict aliasing would change results.
target_compile_options(ppc_recomp PRIVATE -ffp-contract=off -fno-strict-aliasing)
# The console's hardware registers are plain memory in this address space. The
# recompiler marks the stores that address them (eieio follows), and this header
# routes those through a hook so the XMA decoder sees its kick, lock and clear
# registers written. Force-included: ppc/ is regenerated by build.sh.
target_compile_options(ppc_recomp PRIVATE -include "${CMAKE_SOURCE_DIR}/runtime/mmio_hook.h")
option(PES_TRAP_INF "Report stores of infinity (arm with files/trap_inf); slows every store" OFF)
if(PES_TRAP_INF)
    target_compile_definitions(ppc_recomp PRIVATE PES_TRAP_INF=1)
endif()
if(NOT PES17_LOGGING)
    # Compiles every LOGI/LOGW/LOGE/LOGK call to nothing, warnings included.
    add_compile_definitions(PES17_NO_LOGGING=1)
endif()

if(PES17_TRACE_INDIRECT)
    target_compile_definitions(ppc_recomp PUBLIC PES17_TRACE_INDIRECT=1)
endif()

# The glue build.sh generates from the recompiled output -- a stub for every
# kernel import the runtime lacks, a tracing hook for every D3D9 entry point --
# lives beside that output, since it is different for each title.
# ---- online service --------------------------------------------------------
# What the title's network calls reach (runtime/online/service.h): "none" keeps
# system link on this machine, "lan" is plain UDP between machines, "steam" is
# Steam, with lan behind it for when Steam is not running. A backend is the one
# file runtime/online/<name>.cpp; the runtime tries them in service.cpp's order.
set(PES17_ONLINE "none" CACHE STRING "Online service: none, lan or steam")
if(PES17_ONLINE STREQUAL "none")
    set(PES17_ONLINE_BACKENDS "")
elseif(PES17_ONLINE STREQUAL "lan")
    set(PES17_ONLINE_BACKENDS lan)
elseif(PES17_ONLINE STREQUAL "steam")
    set(PES17_ONLINE_BACKENDS steam lan)
else()
    message(FATAL_ERROR "PES17_ONLINE=${PES17_ONLINE}: none, lan or steam")
endif()
set(PES17_ONLINE_SOURCES "")
set(PES17_ONLINE_DEFINITIONS "")
foreach(backend ${PES17_ONLINE_BACKENDS})
    list(APPEND PES17_ONLINE_SOURCES "${CMAKE_SOURCE_DIR}/runtime/online/${backend}.cpp")
    string(TOUPPER "${backend}" backend_upper)
    list(APPEND PES17_ONLINE_DEFINITIONS "PES17_ONLINE_${backend_upper}=1")
endforeach()
message(STATUS "Online service: ${PES17_ONLINE}")

# The runtime checks the disc's executable against this hash before installing.
# The selected local executable is PES 2017 Europe XEX v0x12. Validate its
# execution-info metadata whenever the private input is present; the SHA-256
# below then binds installation to the exact same bytes used for recompilation.
set(PES17_EXPECTED_XEX_VERSION "0x00000012" CACHE STRING "Expected PES 2017 XEX execution-info version")
set(PES17_EXPECTED_TITLE_ID "0x4A3007D1" CACHE STRING "Expected PES 2017 Xbox 360 title ID")
if(EXISTS "${CMAKE_SOURCE_DIR}/pes/default.xex")
    if(NOT EXISTS "${PES17_PPC_DIR}/XEX_TARGET")
        message(FATAL_ERROR "ppc/XEX_TARGET is missing; regenerate PPC output with ./build.sh for the selected XEX")
    endif()
    file(STRINGS "${PES17_PPC_DIR}/XEX_TARGET" PES17_PPC_TARGET_LINES)
    set(PES17_PPC_TARGET_TITLE_ID "")
    set(PES17_PPC_TARGET_VERSION "")
    set(PES17_PPC_TARGET_SHA256 "")
    foreach(PES17_PPC_TARGET_LINE IN LISTS PES17_PPC_TARGET_LINES)
        if(PES17_PPC_TARGET_LINE MATCHES "^title_id=(.*)$")
            set(PES17_PPC_TARGET_TITLE_ID "${CMAKE_MATCH_1}")
        elseif(PES17_PPC_TARGET_LINE MATCHES "^version=(.*)$")
            set(PES17_PPC_TARGET_VERSION "${CMAKE_MATCH_1}")
        elseif(PES17_PPC_TARGET_LINE MATCHES "^sha256=([0-9a-fA-F]+)$")
            set(PES17_PPC_TARGET_SHA256 "${CMAKE_MATCH_1}")
        endif()
    endforeach()
    file(SHA256 "${CMAKE_SOURCE_DIR}/pes/default.xex" PES17_INPUT_XEX_SHA256)
    if(NOT PES17_PPC_TARGET_TITLE_ID STREQUAL PES17_EXPECTED_TITLE_ID OR
       NOT PES17_PPC_TARGET_VERSION STREQUAL PES17_EXPECTED_XEX_VERSION OR
       NOT PES17_PPC_TARGET_SHA256 STREQUAL PES17_INPUT_XEX_SHA256 OR
       (PES17_XEX_SHA256_SP AND NOT PES17_XEX_SHA256_SP STREQUAL PES17_INPUT_XEX_SHA256))
        message(FATAL_ERROR "Generated ppc/ does not match pes/default.xex (target version=${PES17_PPC_TARGET_VERSION}, input SHA-256=${PES17_INPUT_XEX_SHA256}); rerun ./build.sh with the selected XEX")
    endif()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    execute_process(
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/verify_xex.py"
            "${CMAKE_SOURCE_DIR}/pes/default.xex"
            --title-id "${PES17_EXPECTED_TITLE_ID}"
            --version "${PES17_EXPECTED_XEX_VERSION}"
        RESULT_VARIABLE PES17_XEX_VERIFY_RESULT
        OUTPUT_VARIABLE PES17_XEX_VERIFY_OUTPUT
        ERROR_VARIABLE PES17_XEX_VERIFY_ERROR)
    if(NOT PES17_XEX_VERIFY_RESULT EQUAL 0)
        message(FATAL_ERROR "PES 2017 XEX target validation failed:\n${PES17_XEX_VERIFY_OUTPUT}\n${PES17_XEX_VERIFY_ERROR}")
    endif()
    message(STATUS "${PES17_XEX_VERIFY_OUTPUT}")
endif()

# ---- launcher --------------------------------------------------------------
# What a player opens (launcher/): it installs the game and starts it. One
# executable with nothing beside it, like the game's: Dear ImGui on SDL's own
# renderer, the fonts as arrays, and XenonRecomp's patcher, which applies the
# title update, compiled in from its checkout.
if(PES17_USE_SDL)
    enable_language(C)
    FetchContent_Declare(imgui
        GIT_REPOSITORY https://github.com/ocornut/imgui.git
        GIT_TAG v1.91.9b
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(imgui)

    # third_party/fonts/*.ttf as C arrays.
    set(PES17_FONTS_SOURCE "${CMAKE_BINARY_DIR}/launcher_fonts.cpp")
    set(fonts_text "// Generated by CMake from third_party/fonts/.\n")
    foreach(font "Menu=BarlowCondensed-Regular" "Heading=BarlowCondensed-Medium" "Body=BarlowSemiCondensed-Regular")
        string(REPLACE "=" ";" font "${font}")
        list(GET font 0 font_name)
        list(GET font 1 font_file)
        set(font_path "${CMAKE_SOURCE_DIR}/third_party/fonts/${font_file}.ttf")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${font_path}")
        file(READ "${font_path}" font_hex HEX)
        string(LENGTH "${font_hex}" font_length)
        math(EXPR font_size "${font_length} / 2")
        string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," font_bytes "${font_hex}")
        string(APPEND fonts_text "extern const unsigned char kFont${font_name}[] = { ${font_bytes} };\n"
                                 "extern const unsigned kFont${font_name}Size = ${font_size};\n")
    endforeach()
    # The window's icon the same way (tools/make_artwork.py makes it).
    set(icon_path "${CMAKE_SOURCE_DIR}/launcher/icon/launcher.bmp")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${icon_path}")
    file(READ "${icon_path}" icon_hex HEX)
    string(LENGTH "${icon_hex}" icon_length)
    math(EXPR icon_size "${icon_length} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," icon_bytes "${icon_hex}")
    string(APPEND fonts_text "extern const unsigned char kIcon[] = { ${icon_bytes} };\n"
                             "extern const unsigned kIconSize = ${icon_size};\n")
    file(GENERATE OUTPUT "${PES17_FONTS_SOURCE}" CONTENT "${fonts_text}")

    # The game draws its sign-in screen (runtime/signin.cpp) with the same
    # fonts, rasterised by the stb_truetype that comes with Dear ImGui.
    target_sources(pes17 PRIVATE "${PES17_FONTS_SOURCE}")
    target_include_directories(pes17 PRIVATE "${imgui_SOURCE_DIR}")
    target_compile_definitions(pes17 PRIVATE PES17_SIGNIN_PICTURE=1)

    set(XENON "${CMAKE_SOURCE_DIR}/third_party/XenonRecomp")
    file(GLOB LAUNCHER_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/launcher/*.cpp")
    add_executable(launcher ${LAUNCHER_SOURCES}
        "${PES17_FONTS_SOURCE}"
        "${CMAKE_SOURCE_DIR}/runtime/install/crypto.cpp"
        "${CMAKE_SOURCE_DIR}/runtime/install/xex.cpp"
        "${CMAKE_SOURCE_DIR}/runtime/install/spawn.cpp"
        "${imgui_SOURCE_DIR}/imgui.cpp"
        "${imgui_SOURCE_DIR}/imgui_draw.cpp"
        "${imgui_SOURCE_DIR}/imgui_tables.cpp"
        "${imgui_SOURCE_DIR}/imgui_widgets.cpp"
        "${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp"
        "${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer3.cpp"
        "${XENON}/XenonUtils/xex_patcher.cpp"
        "${XENON}/XenonUtils/memory_mapped_file.cpp"
        "${XENON}/thirdparty/libmspack/libmspack/mspack/lzxd.c"
        "${XENON}/thirdparty/tiny-AES-c/aes.c")
    set_target_properties(launcher PROPERTIES OUTPUT_NAME pes17-launcher)
    target_include_directories(launcher PRIVATE
        "${imgui_SOURCE_DIR}" "${imgui_SOURCE_DIR}/backends"
        "${XENON}/XenonUtils" "${XENON}/thirdparty/simde"
        "${XENON}/thirdparty/libmspack/libmspack/mspack"
        "${XENON}/thirdparty/tiny-AES-c" "${XENON}/thirdparty/TinySHA1")
    target_compile_definitions(launcher PRIVATE ${PES17_HASH_DEFINITIONS} NOMINMAX)
    # What release this is, for the launcher to look for a newer one: the
    # release workflow gives the tag. A build without one does not update.
    set(PES17_RELEASE_TAG "" CACHE STRING "The release tag this build is published as, e.g. v0.3.0")
    if(WIN32)
        set(PES17_RELEASE_KIND "windows-${PES17_ONLINE}")
    else()
        set(PES17_RELEASE_KIND "linux-${PES17_ONLINE}")
    endif()
    set_source_files_properties("${CMAKE_SOURCE_DIR}/launcher/update.cpp" PROPERTIES COMPILE_DEFINITIONS
        "PES17_RELEASE_TAG=\"${PES17_RELEASE_TAG}\";PES17_RELEASE_KIND=\"${PES17_RELEASE_KIND}\"")
    # The decoder the game has for its sound, for the menu sounds (launcher/sound.cpp).
    target_link_libraries(launcher PRIVATE SDL3::SDL3-static ffmpeg_xma)
    if(WIN32)
        # The program's icon, as Explorer and the taskbar show it.
        enable_language(RC)
        target_sources(launcher PRIVATE "${CMAKE_SOURCE_DIR}/launcher/icon/launcher.rc")
        target_include_directories(launcher PRIVATE "${CMAKE_SOURCE_DIR}/launcher/compat")
        target_link_libraries(launcher PRIVATE urlmon)
        target_link_options(launcher PRIVATE -static)
        if(NOT PES17_DIAGNOSTICS)
            target_link_options(launcher PRIVATE -mwindows)
        endif()
    else()
        target_link_libraries(launcher PRIVATE pthread)
        if(PES17_PORTABLE)
            target_link_options(launcher PRIVATE -static-libstdc++ -static-libgcc)
        endif()
    endif()
endif()
